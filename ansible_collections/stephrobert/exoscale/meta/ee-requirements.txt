# Paquets Python dont les modules de cette collection ont besoin à
# l'exécution, côté contrôleur.
#
# La borne basse est celle que le runtime emploie : `Client.wait` avec
# `max_wait_time`, présent depuis la 0.16 du SDK. C'est la même borne que
# `pyproject.toml` déclare pour les tests, et un test exige que le SDK installé
# expose chaque méthode qu'un module appelle.
exoscale>=0.16

# Borne posée pour une vulnérabilité, pas pour une fonctionnalité.
#
# `idna` est une dépendance transitive du SDK, par `requests`. Sans borne, un
# résolveur prend la plus ancienne version qui satisfait les contraintes, et
# OSV-Scanner l'a signalé dès sa première exécution sur ce fichier : cette
# résolution est vulnérable à CVE-2026-45409 (GHSA-65pc-fj4g-8rjx), corrigée
# en 3.15. Même constat, même réponse que collection-scaleway.
idna>=3.15
